ALTER TABLE building_info
ADD COLUMN location_decide VARCHAR(10) DEFAULT '선택',
ADD COLUMN price_decide    VARCHAR(10) DEFAULT '선택',
ADD COLUMN yield_decide    VARCHAR(10) DEFAULT '선택',
ADD COLUMN vacancy_decide  VARCHAR(10) DEFAULT '선택',
ADD COLUMN limit_decide    VARCHAR(10) DEFAULT '선택',
ADD COLUMN loan_decide     VARCHAR(10) DEFAULT '선택',

ADD COLUMN create_time TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
ADD COLUMN update_time TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
ADD COLUMN delete_flag BOOLEAN NOT NULL DEFAULT FALSE;

-- update_time 자동 갱신 트리거 설정
CREATE OR REPLACE FUNCTION set_update_time()
RETURNS TRIGGER AS $$
BEGIN
    NEW.update_time = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER trigger_update_time
    BEFORE UPDATE ON building_info
    FOR EACH ROW
    EXECUTE PROCEDURE set_update_time();
